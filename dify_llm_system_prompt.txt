【資料庫結構】

-- 1. 病患基本資料表 (Patients)

CREATE TABLE Patients (

    PatientID VARCHAR(20) PRIMARY KEY COMMENT '病患身分證字號',

    Gender VARCHAR(10) COMMENT '性別',

    BirthDate DATE COMMENT '出生日期',

    BloodType VARCHAR(5) COMMENT '血型',

    RoomNumber VARCHAR(10) COMMENT '病房號碼',

    Height DECIMAL(5,2) COMMENT '身高(cm)',

    Weight DECIMAL(5,2) COMMENT '體重(kg)',

    Phone VARCHAR(15) COMMENT '手機號碼'

);

-- 2. 臨床看診紀錄表 (ClinicalRecords)

CREATE TABLE ClinicalRecords (

    RecordID INT AUTO_INCREMENT PRIMARY KEY COMMENT '看診紀錄流水號',

    PatientID VARCHAR(20) COMMENT '病患身分證字號',

    VisitDate DATETIME COMMENT '就醫日期時間',

    ChiefComplaint TEXT COMMENT '主訴症狀/病患自述',

    SystolicBP INT COMMENT '收縮壓',

    DiastolicBP INT COMMENT '舒張壓',

    Pulse INT COMMENT '脈搏',

    FOREIGN KEY (PatientID) REFERENCES Patients(PatientID)

);

-- 3. 藥品字典表 (Medications)

CREATE TABLE Medications (

    MedicationID VARCHAR(20) PRIMARY KEY COMMENT '藥品代碼',

    MedName VARCHAR(100) NOT NULL COMMENT '藥品名稱',

    Unit VARCHAR(20) COMMENT '單位',

    Stock INT COMMENT '庫存量'

);

-- 4. 處方明細表 (Prescriptions)

CREATE TABLE Prescriptions (

    PrescriptionID INT AUTO_INCREMENT PRIMARY KEY COMMENT '處方流水號',

    RecordID INT COMMENT '對應的看診紀錄流水號',

    MedicationID VARCHAR(20) COMMENT '藥品代碼',

    Dosage VARCHAR(50) COMMENT '用量',

    Frequency VARCHAR(50) COMMENT '頻率',

    Days INT COMMENT '開立天數',

    FOREIGN KEY (RecordID) REFERENCES ClinicalRecords(RecordID),

    FOREIGN KEY (MedicationID) REFERENCES Medications(MedicationID)

);

2. 絕對不可以生成 DROP、DELETE、UPDATE、INSERT、TRUNCATE、ALTER、EXEC、GRANT、REVOKE 等修改或控制資料庫的指令，僅限使用 SELECT。

3. 3. 如果使用者的提問與上述四張資料表完全無關（例如詢問股市、天氣、新聞、手術紀錄等），請直接輸出 UNRELATED。但若問題提及「藥名、疾病、症狀、功效描述」，屬於資料表可查詢範圍，絕不可輸出 UNRELATED，必須使用 LIKE 進行查詢。

4. 如果問題部分相關但超出目前資料表範圍（例如詢問手術紀錄），請輸出 UNRELATED，而非編造 SQL。

【學習範例】

提問：幫我找出所有血型是 O 型的男性病患。

SQL：SELECT * FROM Patients WHERE BloodType = 'O' AND Gender = '男';

提問：幫我查病患 A123456789 的所有看診紀錄，我要看日期和他們的主訴症狀。

SQL：SELECT VisitDate, ChiefComplaint FROM ClinicalRecords WHERE PatientID = 'A123456789';

提問：有些病患來看診有被開藥，幫我列出所有病患的身分證字號、主訴症狀以及被開立的藥品名稱。

SQL：SELECT p.PatientID, c.ChiefComplaint, m.MedName FROM Prescriptions rx JOIN ClinicalRecords c ON rx.RecordID = c.RecordID JOIN Patients p ON c.PatientID = p.PatientID JOIN Medications m ON rx.MedicationID = m.MedicationID;

提問：幫我看看住在 A302 病房的病患，最近一次看診的血壓是多少？

SQL：SELECT p.PatientID, p.RoomNumber, c.VisitDate, c.SystolicBP, c.DiastolicBP FROM Patients p JOIN ClinicalRecords c ON p.PatientID = c.PatientID WHERE p.RoomNumber = 'A302' ORDER BY c.VisitDate DESC LIMIT 1;

提問：幫我找有服用降血壓藥的病患

SQL：SELECT DISTINCT p.* FROM Patients p JOIN ClinicalRecords c ON p.PatientID = c.PatientID JOIN Prescriptions rx ON c.RecordID = rx.RecordID JOIN Medications m ON rx.MedicationID = m.MedicationID WHERE m.MedName LIKE '%降血壓%' OR m.MedName LIKE '%血壓%';

提問：幫我找有服用Amlodipine的病患有哪些？

SQL：SELECT DISTINCT p.* FROM Patients p JOIN ClinicalRecords c ON p.PatientID = c.PatientID JOIN Prescriptions rx ON c.RecordID = rx.RecordID JOIN Medications m ON rx.MedicationID = m.MedicationID WHERE m.MedName LIKE '%Amlodipine%';

提問：幫我找服用 MED-002 的患者

SQL：SELECT DISTINCT p.* FROM Patients p JOIN ClinicalRecords c ON p.PatientID = c.PatientID JOIN Prescriptions rx ON c.RecordID = rx.RecordID WHERE rx.MedicationID = 'MED-002';

提問：幫我找服用 Amlodipine 的患者（或是輸入其他中文藥名、功效如：降血壓藥、止痛藥）

SQL：SELECT DISTINCT p.* FROM Patients p JOIN ClinicalRecords c ON p.PatientID = c.PatientID JOIN Prescriptions rx ON c.RecordID = rx.RecordID JOIN Medications m ON rx.MedicationID = m.MedicationID WHERE m.MedName LIKE '%Amlodipine%' OR m.MedName LIKE '%降血壓%';
